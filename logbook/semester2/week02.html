Week 02: Small plasmids recovery issue (once more..)

Overview

  • The minOverlap value plays a key role in small plasmid recovery during Flye assembly, as default values are often higher than typical small plasmid lengths (<2000 bp).

  • Why the 100x seqkit-dorado (SUP) dataset still missed the small plasmid can be explained by:

    • Dorado performing unnecessary trimming on plasmid reads (specifically, fully spanning reads). Combined with high read depth/coverage, this introduced unnecessary noise during Flye’s disjointig generation step.
    • High coverage + read trimming increasing inner overlap relationships in disjointig generation.
    • Multithreading also influencing disjointig generation and filtering, causing small plasmids to be dropped.

  • Lowering the minOverlap value had no negative impact, at least across the parameters assessed.

So, the causes are:

  • First, the minOverlap parameter plays an important role in disjointig generation and later in repeat graph collapsing.

  • Second, trimming has a negative impact on reads that are important for small plasmid recovery, especially on fully spanning reads in high-depth datasets.

  • Third, Flye produces non-deterministic results, where multithreading causes disjointig removal and impact on small plasmid recovery.

print("Reading and merging data...")

# 2. Load data and tag with minOverlap values
df_default = pd.read_csv(csv_default)
df_default["minOverlap"] = "default"

df_1000 = pd.read_csv(csv_1000)
df_1000["minOverlap"] = "1000"

df = pd.concat([df_default, df_1000], ignore_index=True)

# 3. Colourblind-friendly palette from colour universal design (CUD)
named_colors = {
    "black": "#000000",
    "orange": "#e69f00",
    "skyblue": "#56b4e9",
    "vermilion": "#d55e00",
    "bluish green": "#009e73",
    "yellow": "#f0e442",
    "blue": "#0072b2",
    "reddish purple": "#cc79a7",
}
cud_palette = list(named_colors.values())

def cud(n: int = len(cud_palette), start: int = 0) -> List[str]:
    remainder = cud_palette[:start]
    palette = cud_palette[start:] + remainder
    return palette[:n]

# 4. Plot settings
sns.set_theme(style="whitegrid")
models = ["sup", "hac"]
depths = ["100x", "50x", "20x"]
hue_order = ["default", "1000"] # minOverlap is now our hue

estimators = ["mean", "median"]
scales = ["linear", "logit"]

# Sort tools (combos) alphabetically as requested
order_abs = sorted(df["combo"].unique())

for est in estimators:
    if est == "mean":
        est_func = np.mean
    else:
        est_func = np.median

    for p_type in plot_types:
        for scale in scales: 
            # Create a 3x2 grid: 3 depths (rows), 2 models (columns)
            fig, axes = plt.subplots(nrows=len(depths), ncols=len(models), 
                                     figsize=(14, 12), dpi=300, sharex=True, sharey=False)

            for r, depth in enumerate(depths):
                for c, model in enumerate(models):
                    ax = axes[r, c]
                    # Filter data for this specific facet
                    df_sub = df.query("model == @model and depth == @depth").copy()

                    if not df_sub.empty:
                        cap = 0.9999
                        
                        # Apply logit cap ONLY if plotting logit scale
                        if scale == "logit":
                            df_sub['NGA50_norm'] = df_sub['NGA50_norm'].apply(
                                lambda v: cap if v >= cap else (0.00001 if v <= 0 else v)
                            )

                        if p_type == "pointplot":
                            sns.pointplot(
                                data=df_sub, x="combo", y="NGA50_norm", hue="minOverlap",
                                order=order_abs, hue_order=hue_order,
                                palette=cud(len(hue_order), start=2),

                                err_kws={'linewidth': 1.5}, estimator=est_func
                            )
                        elif p_type == "stripplot":
                            sns.boxplot(
                                data=df_sub, x="combo", y="NGA50_norm", hue="minOverlap",
                                order=order_abs, hue_order=hue_order,
                                palette=cud(len(hue_order), start=2),
                                ax=ax, fliersize=0, gap=0.1
                            )
                            sns.stripplot(
                                data=df_sub, x="combo", y="NGA50_norm", hue="minOverlap",
                                order=order_abs, hue_order=hue_order,
                                palette=cud(len(hue_order), start=2),
                                ax=ax, alpha=0.6, dodge=True, linewidth=0.5, edgecolor="black", legend=False
                            )

                    # Dynamic Axis Scaling and Zoom
                    if scale == "logit":
                        ax.set_yscale("logit", nonpositive="clip")
                        ax.set_ylim(0.5, 0.999)
                        
                        # Custom ticks strictly for the zoomed logit region
                        yticks = [0.5, 0.6, 0.7, 0.8, 0.9, 0.99, 0.999, cap]
                        yticklabels = [f"{yval:.2%}" for yval in yticks]
                        ax.set_yticks(yticks)
                        ax.set_yticklabels(yticklabels)
                    elif scale == "linear":
                        ax.set_yscale("linear")
                        ax.set_ylim(-0.05, 1.05) 

                    # Text Formatting & Layout Management
                    # Only show y-axis labels on the leftmost column
                    if c == 0:
                        ax.set_ylabel(f"NGA50 (Norm) [{scale.title()} | {est.title()}]")
                    else:
                        ax.set_ylabel("")
                        
                    ax.set_title(f"Model: {model} | Depth: {depth}")
                    ax.set_xlabel("")

                    # Only show x-axis tick labels on the bottom row
                    if r == len(depths) - 1:
                        ax.set_xticks(range(len(order_abs)))
                        ax.set_xticklabels(order_abs, rotation=45, ha="right", rotation_mode="anchor", fontsize=11)

                    elif ax.get_legend() is not None:
                        ax.get_legend().remove()

            fig.tight_layout()

            # Save the figure
            new_output_path = out_dir / f"combo_assembly_nga50_normalised_3x2_{scale}_{p_type}_{est}.png"
            fig.savefig(new_output_path, bbox_inches='tight')
            plt.close(fig)
            print(f"Saved {new_output_path.name}")

df_default = pd.read_csv(csv_default)

df_1000["minOverlap"] = "1000"

# Combine both datasets
df = pd.concat([df_default, df_1000], ignore_index=True)

# 3. Colourblind-friendly palette from colour universal design (CUD)
named_colors = {
    "black": "#000000",
    "orange": "#e69f00",
    "skyblue": "#56b4e9",
    "vermilion": "#d55e00",
    "bluish green": "#009e73",
    "yellow": "#f0e442",
    "blue": "#0072b2",
    "reddish purple": "#cc79a7",
}
cud_palette = list(named_colors.values())

def cud(n: int = len(cud_palette), start: int = 0) -> List[str]:
    remainder = cud_palette[:start]
    palette = cud_palette[start:] + remainder
    return palette[:n]

# 4. Plotting preparation
sns.set_theme(style="whitegrid")
models = ["sup", "hac"]
depths = ["100x", "50x", "20x"]
hue_order = ["default", "1000"]

order_abs = sorted(df["combo"].unique())

fig, axes = plt.subplots(nrows=len(depths), ncols=len(models), 
                         figsize=(14, 12), dpi=300, sharex=True, sharey=True)

for r, depth in enumerate(depths):
    for c, model in enumerate(models):
        ax = axes[r, c]
        
        # Filter for the specific facet
        df_sub = df.query("model == @model and depth == @depth").copy()

        if not df_sub.empty:
            # Barplot using estimator=sum to calculate the total across all samples
            sns.barplot(
                data=df_sub, x="combo", y="contamination_count", hue="minOverlap", 
                order=order_abs, hue_order=hue_order,
                palette=cud(len(hue_order), start=2), 
                ax=ax, dodge=True, edgecolor="black", linewidth=0.5,

            )
